#!/usr/bin/env python3
"""Build dist/index.html (one self-contained file) from src/.

    python3 tools/build.py            # writes dist/index.html
    python3 tools/build.py --check    # exit 1 if any module has a JS syntax error (needs node)

No dependencies beyond Python 3. See docs/ARCHITECTURE.md.
"""
import hashlib, json, re, shutil, subprocess, sys, tempfile
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
SRC = ROOT / "src"


def read(p):
    return Path(p).read_text(encoding="utf-8")


def js_bundle():
    parts = []
    for raw in read(SRC / "js" / "manifest.txt").splitlines():
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        if line.startswith("data"):
            head, name, rel = line.split()
            f = ROOT / rel
            if head == "data?" and not f.exists():      # optional data (e.g. your own LEE file): absent -> null
                parts.append(f"/* ---- data: {rel} (absent) ---- */\nconst {name}=null;\n")
                continue
            obj = json.loads(read(f))                   # validates the JSON
            parts.append(f"/* ---- data: {rel} ---- */\nconst {name}={json.dumps(obj, separators=(',', ':'))};\n")
        else:
            parts.append(f"/* ---- {line} ---- */\n" + read(SRC / "js" / line).rstrip() + "\n")
    return "\n".join(parts)


def css_bundle():
    return "\n".join(read(p).rstrip() for p in sorted((SRC / "css").glob("*.css"), key=lambda p: (p.name != "base.css", p.name)))


def build():
    html = read(SRC / "index.html")
    html = re.sub(r"<!--@include (.+?)-->", lambda m: read(SRC / m.group(1)).rstrip(), html)
    html = html.replace("<!--@css-->", css_bundle()).replace("<!--@js-->", js_bundle())
    return html


def standalone(fragment):
    """The template is a fragment (title, styles, body) so it can also be pasted into hosts that add their own
    document skeleton. For hosting on its own it needs a doctype, <html>, <head> and <body>."""
    i = fragment.index("</style>") + len("</style>")
    return ('<!doctype html>\n<html lang="en">\n<head>\n<meta charset="utf-8">\n' + fragment[:i] +
            '\n</head>\n<body>' + fragment[i:] + '\n</body>\n</html>\n')


def main():
    html = build()
    if "--check" in sys.argv:
        for i, s in enumerate(re.findall(r"<script>(.*?)</script>", html, re.S)):
            with tempfile.NamedTemporaryFile("w", suffix=".js", delete=False, encoding="utf-8") as f:
                f.write(s)
            r = subprocess.run(["node", "--check", f.name], capture_output=True, text=True)
            if r.returncode:
                print(r.stderr); sys.exit(1)
        print("syntax ok")
        return
    dist = ROOT / "dist"
    dist.mkdir(exist_ok=True)
    page = standalone(html)
    (dist / "index.html").write_text(page, encoding="utf-8")
    # web-app files: manifest, icons, service worker (cache name carries a hash of the page so updates refresh it)
    shutil.copy(SRC / "pwa" / "manifest.webmanifest", dist / "manifest.webmanifest")
    shutil.copytree(SRC / "pwa" / "icons", dist / "icons", dirs_exist_ok=True)
    version = hashlib.sha256(page.encode("utf-8")).hexdigest()[:10]
    (dist / "sw.js").write_text(read(SRC / "pwa" / "sw.js").replace("__VERSION__", version), encoding="utf-8")
    print(f"wrote dist/ (index.html {len(page)/1024:.0f} KB, app files, cache version {version})")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Make tests/reference-quality.json: CRI / TM-30 values from colour-science for a set of spectra, so tests/maths.mjs can
check the JavaScript implementation. Dev tool (needs `pip install colour-science`). Optional extra spectra: pass CSV files
(wavelength, intensity) as arguments - they are used as extra test cases.

    python3 tools/gen_quality_reference.py [led1.csv led2.csv ...]
"""
import json, sys, numpy as np, colour
from pathlib import Path
from colour import SpectralDistribution, SpectralShape, SDS_ILLUMINANTS, sd_blackbody
from colour.quality import colour_rendering_index, colour_fidelity_index_CIE2017

shape = SpectralShape(380, 780, 5)
cases = {}
for name in ['FL1','FL2','FL3','FL4','FL5','FL6','FL7','FL8','FL9','FL10','FL11','FL12','FL3.1','FL3.5','FL3.10','FL3.15',
             'LED-B1','LED-B2','LED-B3','LED-B4','LED-B5','LED-BH1','LED-RGB1','LED-V1','LED-V2','D65','A']:
    if name in SDS_ILLUMINANTS:
        cases[name] = SDS_ILLUMINANTS[name].copy().align(shape).values
for T in (2200, 2700, 3000, 3500, 4000, 4500, 5000, 5500, 6500, 10000):
    cases[f'Planck {T}'] = sd_blackbody(T, shape).values
w = shape.wavelengths
g = lambda m, s: np.exp(-0.5 * ((w - m) / s) ** 2)
cases['gauss RGB'] = g(455, 10) + 0.9 * g(530, 18) + 1.1 * g(630, 10)
cases['pc-white 5600'] = 1.0 * g(450, 9) + 0.55 * g(560, 55) + 0.25 * g(620, 45)
cases['pc-white 2700'] = 0.5 * g(450, 9) + 0.55 * g(580, 60) + 0.55 * g(640, 45)
for f in sys.argv[1:]:
    rows = [l.split(',') for l in Path(f).read_text(encoding='utf-8', errors='ignore').splitlines()]
    d = np.array([[float(a), float(b)] for a, b, *_ in (r for r in rows if len(r) > 1 and r[0].replace('.', '', 1).isdigit())])
    cases['file ' + Path(f).stem] = np.interp(w, d[:, 0], d[:, 1], left=0, right=0)

out = []
for name, v in cases.items():
    v = np.asarray(v, float); v = v / v.max()
    sd = SpectralDistribution(v, w, name=name)
    rec = {'name': name, 'spd5': [round(float(x), 7) for x in v]}
    try:
        f = colour_fidelity_index_CIE2017(sd, additional_data=True)
        rec.update(cct=float(f.CCT), duv=float(f.D_uv), Rf=float(f.R_f), Rfi=[float(x) for x in f.R_s])
        # bin statistics (same formulas as the JS, from colour-science's own CAM02-UCS values)
        t, r = f.colorimetry_data
        ap_t, bp_t = t.Jpapbp[:, 1], t.Jpapbp[:, 2]; ap_r, bp_r = r.Jpapbp[:, 1], r.Jpapbp[:, 2]
        hue = r.JMh[:, 2]; b = np.minimum(15, np.floor(hue / 22.5).astype(int))
        m = lambda a: np.array([a[b == k].mean() for k in range(16)])
        at, bt, ar, br = m(ap_t), m(bp_t), m(ap_r), m(bp_r)
        area = lambda x, y: abs(sum(x[i] * y[(i + 1) % 16] - x[(i + 1) % 16] * y[i] for i in range(16))) / 2
        rec['Rg'] = float(100 * area(at, bt) / area(ar, br))
        rec['Rcs'] = [float(x) for x in (np.hypot(at, bt) - np.hypot(ar, br)) / np.hypot(ar, br)]
    except Exception as e:
        rec['fid_error'] = str(e)
    try:
        c = colour_rendering_index(sd, additional_data=True)
        rec.update(Ra=float(c.Q_a), R=[float(c.Q_as[i].Q_a) for i in range(1, 15)])
    except Exception as e:
        rec['cri_error'] = str(e)
    out.append(rec)
Path(__file__).resolve().parent.parent.joinpath('tests', 'reference-quality.json').write_text(json.dumps(out).replace('NaN', 'null'))
print(len(out), 'cases;', sum('Ra' in r for r in out), 'with CRI;', sum('Rf' in r for r in out), 'with TM-30')

#!/usr/bin/env python3
"""Convert the scraper output (spectra_long.csv + metadata.csv) into data/lee.json.

    python3 tools/lee_scraper.py                # writes lee_output/
    python3 tools/lee_csv_to_json.py lee_output/spectra_long.csv lee_output/metadata.csv -o data/lee.json

Output: {"w":[405..700 step 5], "f":[[id, name, [percent or null per wavelength]], ...]}
"""
import argparse, collections, csv, json
ap = argparse.ArgumentParser()
ap.add_argument("long_csv"); ap.add_argument("metadata_csv"); ap.add_argument("-o", "--out", default="data/lee.json")
ap.add_argument("--start", type=int, default=405); ap.add_argument("--stop", type=int, default=700); ap.add_argument("--step", type=int, default=5)
a = ap.parse_args()
meta = collections.OrderedDict((r["filter_id"], r["name"]) for r in csv.DictReader(open(a.metadata_csv, encoding="utf-8-sig")))
spec = collections.defaultdict(dict)
for r in csv.DictReader(open(a.long_csv, encoding="utf-8-sig")):
    spec[r["filter_id"]][int(float(r["wavelength_nm"]))] = round(float(r["transmission_percent"]), 3)
grid = list(range(a.start, a.stop + 1, a.step))
data = {"w": grid, "f": [[fid, meta[fid], [spec[fid].get(w) for w in grid]] for fid in meta if fid in spec]}
open(a.out, "w", encoding="utf-8").write(json.dumps(data, separators=(",", ":"), ensure_ascii=False) + "\n")
print(f"{len(data['f'])} filters -> {a.out}")

===== tools/lee_scraper.py =====
#!/usr/bin/env python3
"""
Scrape LEE Filters spectral transmission curves into CSV files.

Each filter page (e.g. https://leefilters.com/colour/247-lee-minus-green/) has an
inline SVG chart with one <circle class="tooltip" title="<span>405</span>Colour: <b>70.336</b>">
per wavelength. This script reads those circles, so no JavaScript rendering is needed
(if it turns out to be needed, use --render, see below).

Usage:
    python lee_scraper.py                      # discover filters via sitemap, scrape all
    python lee_scraper.py --urls-file urls.txt # use your own list of URLs
    python lee_scraper.py --limit 5            # test run with 5 filters
    python lee_scraper.py --render             # use Playwright if plain HTTP finds no data

Output (in ./lee_output):
    spectra_long.csv   filter_id, name, wavelength_nm, transmission_percent, transmission
    spectra_wide.csv   wavelength_nm rows x one column per filter (fraction 0..1)
    metadata.csv       filter_id, name, url, n_points, wl_min, wl_max, plus site-published Y/x/y
    spectra/<id>.csv   one small file per filter
    cache/<id>.html    raw pages, so re-runs don't hit the site again
"""
import argparse
import csv
import html
import re
import sys
import time
import urllib.request
import urllib.error
from pathlib import Path

UA = "Mozilla/5.0 (compatible; personal-spectra-scraper/1.0)"
BASE = "https://leefilters.com"
SITEMAP_CANDIDATES = [
    f"{BASE}/sitemap_index.xml",
    f"{BASE}/wp-sitemap.xml",
    f"{BASE}/sitemap.xml",
    f"{BASE}/sitemap/",
]
# Filter pages look like /colour/247-lee-minus-green/
DEFAULT_URL_PATTERN = r"^https://leefilters\.com/colour/[^/]+/?$"

# Tag matcher that respects quoted attribute values, because the title attribute may
# contain raw "<span>...</b>" (unescaped < and >) as well as &lt;-escaped forms.
CIRCLE_RE = re.compile(
    r"<circle\b(?=(?:[^>\"']|\"[^\"]*\"|'[^']*')*?\bclass=[\"']tooltip[\"'])"
    r"(?:[^>\"']|\"[^\"]*\"|'[^']*')*>",
    re.S,
)
TITLE_ATTR_RE = re.compile(r"\btitle=(?:\"([^\"]*)\"|'([^']*)')", re.S)
POINT_RE = re.compile(r"<span>\s*([\d.]+)\s*</span>.*?<b>\s*(-?[\d.]+)\s*</b>", re.S)
H1_RE = re.compile(r"<h1[^>]*>(.*?)</h1>", re.S | re.I)


def fetch(url, retries=3, delay=2.0):
    last = None
    for attempt in range(retries):
        try:
            req = urllib.request.Request(url, headers={"User-Agent": UA})
            with urllib.request.urlopen(req, timeout=30) as r:
                return r.read().decode("utf-8", errors="replace")
        except (urllib.error.URLError, TimeoutError) as e:
            last = e
            time.sleep(delay * (attempt + 1))
    raise RuntimeError(f"fetch failed for {url}: {last}")


def discover_urls(pattern):
    """Walk sitemap(s) recursively and keep URLs matching pattern."""
    pat = re.compile(pattern)
    seen_maps, found = set(), set()

    def walk(sm_url):
        if sm_url in seen_maps:
            return
        seen_maps.add(sm_url)
        try:
            text = fetch(sm_url)
        except RuntimeError:
            return
        for loc in re.findall(r"<loc>\s*(.*?)\s*</loc>", text, re.S):
            loc = html.unescape(loc)
            if loc.endswith(".xml"):
                walk(loc)
            elif pat.match(loc):
                found.add(loc)

    for c in SITEMAP_CANDIDATES:
        walk(c)
        if found:
            break
    return sorted(found)


def parse_page(page):
    """Return (name, {wavelength: transmission_percent}, site_values)."""
    m = H1_RE.search(page)
    name = html.unescape(re.sub(r"<[^>]+>", "", m.group(1))).strip() if m else ""

    pts = {}
    for tag in CIRCLE_RE.findall(page):
        t = TITLE_ATTR_RE.search(tag)
        if not t:
            continue
        title = html.unescape(t.group(1) or t.group(2))
        pm = POINT_RE.search(title)
        if pm:
            pts[float(pm.group(1))] = float(pm.group(2))

    # Values the site publishes next to the chart (Daylight/Source C, then Tungsten).
    text = re.sub(r"<[^>]+>", " ", page)
    text = re.sub(r"\s+", " ", html.unescape(text))
    site = {}
    for label, key in (("Transmission Y", "Y"), ("x", "x"), ("y", "y")):
        vals = re.findall(rf"(?:^|\s){re.escape(label)}\s+([\d.]+)", text)
        if len(vals) >= 1:
            site[f"{key}_daylight"] = vals[0]
        if len(vals) >= 2:
            site[f"{key}_tungsten"] = vals[1]
    return name, pts, site


def render_with_playwright(urls_to_html):
    """Fallback: fetch pages with a headless browser (pip install playwright; playwright install chromium)."""
    from playwright.sync_api import sync_playwright

    with sync_playwright() as p:
        browser = p.chromium.launch()
        page = browser.new_page()
        for url in list(urls_to_html):
            page.goto(url, wait_until="networkidle")
            page.wait_for_selector("circle.tooltip", state="attached", timeout=15000)
            urls_to_html[url] = page.content()
        browser.close()


def slug_of(url):
    return url.rstrip("/").rsplit("/", 1)[-1]


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--urls-file", help="text file with one filter URL per line")
    ap.add_argument("--pattern", default=DEFAULT_URL_PATTERN, help="regex for sitemap URLs to keep")
    ap.add_argument("--out", default="lee_output")
    ap.add_argument("--limit", type=int, default=0)
    ap.add_argument("--delay", type=float, default=1.0, help="seconds between requests")
    ap.add_argument("--render", action="store_true", help="use Playwright when plain HTML has no data")
    args = ap.parse_args()

    out = Path(args.out)
    (out / "cache").mkdir(parents=True, exist_ok=True)
    (out / "spectra").mkdir(exist_ok=True)

    if args.urls_file:
        urls = [u.strip() for u in Path(args.urls_file).read_text().splitlines() if u.strip()]
    else:
        print("Discovering filter URLs from sitemap ...")
        urls = discover_urls(args.pattern)
        print(f"  found {len(urls)} URLs")
        if not urls:
            sys.exit("No URLs found. Put URLs in a text file and use --urls-file.")
    if args.limit:
        urls = urls[: args.limit]

    results = {}  # slug -> dict
    failures = []
    for i, url in enumerate(urls, 1):
        slug = slug_of(url)
        cache = out / "cache" / f"{slug}.html"
        if cache.exists():
            page = cache.read_text(encoding="utf-8")
        else:
            try:
                page = fetch(url)
            except RuntimeError as e:
                print(f"[{i}/{len(urls)}] {slug}: {e}")
                failures.append((url, "fetch error"))
                continue
            cache.write_text(page, encoding="utf-8")
            time.sleep(args.delay)

        name, pts, site = parse_page(page)
        if not pts and args.render:
            holder = {url: None}
            render_with_playwright(holder)
            page = holder[url]
            cache.write_text(page, encoding="utf-8")
            name, pts, site = parse_page(page)
        if not pts:
            print(f"[{i}/{len(urls)}] {slug}: no spectral data (not a filter page?)")
            failures.append((url, "no data"))
            continue
        print(f"[{i}/{len(urls)}] {slug}: {len(pts)} points ({min(pts):.0f}-{max(pts):.0f} nm)")
        results[slug] = {"name": name, "url": url, "pts": pts, "site": site}

    if not results:
        sys.exit("Nothing scraped. If pages had no data, try --render.")

    # per-filter files + long CSV
    with open(out / "spectra_long.csv", "w", newline="", encoding="utf-8") as fl:
        w = csv.writer(fl)
        w.writerow(["filter_id", "name", "wavelength_nm", "transmission_percent", "transmission"])
        for slug, r in results.items():
            with open(out / "spectra" / f"{slug}.csv", "w", newline="", encoding="utf-8") as f1:
                w1 = csv.writer(f1)
                w1.writerow(["wavelength_nm", "transmission"])
                for wl in sorted(r["pts"]):
                    v = r["pts"][wl]
                    w.writerow([slug, r["name"], f"{wl:g}", v, round(v / 100, 6)])
                    w1.writerow([f"{wl:g}", round(v / 100, 6)])

    # wide CSV (fractions 0..1)
    all_wl = sorted({wl for r in results.values() for wl in r["pts"]})
    slugs = list(results)
    with open(out / "spectra_wide.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["wavelength_nm"] + slugs)
        for wl in all_wl:
            w.writerow([f"{wl:g}"] + [
                round(results[s]["pts"][wl] / 100, 6) if wl in results[s]["pts"] else "" for s in slugs
            ])

    # metadata
    site_keys = sorted({k for r in results.values() for k in r["site"]})
    with open(out / "metadata.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["filter_id", "name", "url", "n_points", "wl_min", "wl_max"] + site_keys)
        for s, r in results.items():
            w.writerow([s, r["name"], r["url"], len(r["pts"]), f"{min(r['pts']):g}", f"{max(r['pts']):g}"]
                       + [r["site"].get(k, "") for k in site_keys])

    print(f"\nDone: {len(results)} filters -> {out}/")
    if failures:
        print(f"{len(failures)} URLs skipped:")
        for u, why in failures:
            print(f"  {why}: {u}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Write N synthetic white-LED spectra (blue pump + broad phosphor + optional red phosphor) as one CSV per LED, in the same
layout as an instrument export ("# key,value" header, then wavelength,intensity 350-850 nm, 1 nm). For tests, demos and for
trying the LED library with a large set:   python3 tools/make_demo_leds.py 1000 out_dir"""
import sys, numpy as np
from pathlib import Path
n = int(sys.argv[1]) if len(sys.argv) > 1 else 100

for i in range(n):
    pump = g(rng.uniform(438, 465), rng.uniform(7, 12)) * rng.uniform(0.6, 1.2)
    ph_mu, ph_s = rng.uniform(535, 600), rng.uniform(38, 70)
    phos = (g(ph_mu, ph_s) + 0.35 * g(ph_mu + 45, ph_s * 1.2)) * rng.uniform(0.3, 0.9)
    red = g(rng.uniform(615, 660), rng.uniform(8, 35)) * (rng.uniform(0.15, 0.9) if rng.random() < 0.6 else 0)
    v = pump + phos + red + rng.normal(0, 2e-4, w.size) + 1e-4
    v = np.clip(v / v.max(), 0, None)
    (out / f'led_{i:04d}.csv').write_text('# status,normal\n# device,DEMO-%04d\nWavelength (nm),Intensity\n' % i +
        '\n'.join(f'{a},{b:.7f}' for a, b in zip(w, v)) + '\n')
print('wrote', n, 'files to', out)

#!/usr/bin/env python3
"""Generate data/gels.json: ~100 synthetic example colour gels (NOT measurements, NOT any manufacturer's data).

Each gel is a smooth parametric transmission curve (band-pass, long-pass, short-pass, notch, colour-temperature
tilt, neutral density, diffusion). Names are generic descriptions. Deterministic: same output every run.

    python3 tools/make_example_gels.py            # writes data/gels.json
Format: {"w0":380,"s":5,"f":[[id,name,[percent per 5 nm, 380..780]], ...]}
"""
import json, math
from pathlib import Path

W = list(range(380, 781, 5))
sig = lambda x: 1 / (1 + math.exp(-x))
gel = []  # (name, fn)


def add(name, fn):
    gel.append((name, fn))


def band(c, w, tmax, edge=12):      # smooth band-pass, FWHM ~ w
    return lambda l: tmax * sig((l - (c - w / 2)) / edge) * sig(((c + w / 2) - l) / edge)


def longpass(c, tmax, edge=14):
    return lambda l: tmax * sig((l - c) / edge)


def shortpass(c, tmax, edge=14):
    return lambda l: tmax * sig((c - l) / edge)


def notch(c, w, depth, base=0.9, edge=12):
    return lambda l: base * (1 - depth * sig((l - (c - w / 2)) / edge) * sig(((c + w / 2) - l) / edge))


def tilt(k, tmax=0.85):             # colour-temperature style: log-transmission linear in wavelength
    return lambda l: tmax * math.exp(k * (l - 580) / 200)


def nd(t):
    return lambda l: t * (1 + 0.04 * (l - 580) / 200)


# saturated colours: (name, centre, width, tmax)
for nm, c, w, t in [("Deep Violet", 430, 50, .35), ("Royal Blue", 450, 55, .30), ("Deep Blue", 460, 60, .22),
                    ("Sky Blue", 470, 90, .55), ("Light Blue", 480, 140, .70), ("Cyan", 500, 90, .55),
                    ("Teal", 510, 70, .40), ("Blue Green", 520, 80, .45), ("Emerald", 530, 60, .35),
                    ("Primary Green", 535, 55, .45), ("Leaf Green", 545, 70, .50), ("Light Green", 550, 120, .70),
                    ("Lime", 565, 80, .70), ("Yellow Green", 570, 90, .72)]:
    add(nm, band(c, w, t))
for nm, c, t in [("Pale Yellow", 520, .88), ("Light Yellow", 510, .84), ("Medium Yellow", 525, .80), ("Deep Yellow", 535, .74),
                 ("Golden Yellow", 545, .70), ("Amber", 560, .72), ("Light Amber", 545, .80), ("Deep Amber", 575, .68),
                 ("Orange", 585, .72), ("Deep Orange", 595, .66), ("Fire Orange", 605, .66), ("Scarlet", 610, .70),
                 ("Light Red", 620, .72), ("Primary Red", 600, .64), ("Medium Red", 625, .66), ("Deep Red", 635, .55),
                 ("Dark Red", 650, .42)]:
    add(nm, longpass(c, t, 10 if t < .7 else 14))
for nm, c, t in [("Pale Cyan", 600, .80), ("Light Blue Tint", 580, .82), ("Ice Blue", 560, .78), ("Sea Blue", 540, .70),
                 ("Aqua", 560, .68), ("Medium Cyan", 575, .60)]:
    add(nm, shortpass(c, t, 25))
# pinks / magentas / lavenders: notch in green
for nm, c, w, d, b in [("Pale Pink", 540, 90, .20, .90), ("Light Pink", 545, 100, .35, .88), ("Rose", 545, 100, .55, .84),
                       ("Deep Rose", 545, 90, .75, .78), ("Hot Pink", 550, 90, .85, .76), ("Magenta", 550, 100, .93, .74),
                       ("Deep Magenta", 550, 110, .97, .66), ("Purple", 540, 130, .96, .55), ("Light Lavender", 560, 110, .25, .90),
                       ("Lavender", 560, 110, .45, .85), ("Orchid", 555, 110, .70, .78), ("Violet Pink", 545, 120, .85, .70),
                       ("Salmon", 490, 80, .40, .88), ("Peach", 480, 90, .30, .88), ("Coral", 500, 90, .55, .80),
                       ("Minus Green (light)", 545, 90, .30, .90), ("Minus Green (medium)", 545, 90, .55, .88),
                       ("Minus Green (deep)", 545, 90, .80, .86), ("Minus Red", 640, 110, .85, .85),
                       ("Minus Blue", 450, 80, .80, .85)]:
    add(nm, notch(c, w, d, b))
# colour temperature conversion
for i, k in enumerate([-1.6, -1.2, -0.8, -0.5, -0.25], 1):
    add(["Warm Correction Full", "Warm Correction ¾", "Warm Correction ½", "Warm Correction ¼", "Warm Correction ⅛"][i - 1], tilt(k, .78))
for i, k in enumerate([1.6, 1.2, .8, .5, .25]):
    add(["Cool Correction Full", "Cool Correction ¾", "Cool Correction ½", "Cool Correction ¼", "Cool Correction ⅛"][i], tilt(k, .62))
for nm, k in [("Straw Warm", -1.0), ("Sunset Warm", -2.0), ("Candle Warm", -2.6), ("Daylight Blue Tint", 1.0), ("Overcast Cool", 2.0), ("Moonlight Cool", 2.8)]:
    add(nm, tilt(k, .70))
# neutral density, with slight tint
for nm, t in [("Neutral Grey 0.15", .70), ("Neutral Grey 0.3", .50), ("Neutral Grey 0.6", .25), ("Neutral Grey 0.9", .125), ("Neutral Grey 1.2", .063), ("Neutral Grey 1.8", .016)]:
    add(nm, nd(t))
add("Warm Grey", lambda l: .35 * math.exp(-.35 * (l - 580) / 200))
add("Cool Grey", lambda l: .35 * math.exp(.35 * (l - 580) / 200))
# diffusion / frost: nearly clear, mild scatter loss at the blue end
for nm, t, k in [("Light Frost", .92, .15), ("Medium Frost", .85, .20), ("Heavy Frost", .72, .25), ("Opal Diffusion", .62, .30), ("Clear", .94, .0)]:
    add(nm, lambda l, t=t, k=k: t * math.exp(k * (l - 580) / 400))
# special: twin-band / dichroic-like
add("Dichroic Blue", lambda l: .85 * sig((500 - l) / 8) + .08)
add("Dichroic Amber", lambda l: .85 * sig((l - 565) / 8) + .05)
add("Dichroic Magenta", lambda l: .85 * (1 - sig((l - 505) / 8) * sig((585 - l) / 8)) + .02)
add("Dichroic Green", band(540, 70, .8, 7))
add("Dichroic Cyan", lambda l: .85 * sig((590 - l) / 8) + .05)
add("Twin Peak Blue-Red", lambda l: .6 * (math.exp(-((l - 450) / 25) ** 2) + math.exp(-((l - 640) / 30) ** 2)))
add("Twin Peak Green-Violet", lambda l: .55 * (math.exp(-((l - 420) / 25) ** 2) + math.exp(-((l - 540) / 28) ** 2)))
add("UV Block Clear", longpass(405, .92, 6))
add("Wide Cyan Green", band(510, 150, .65))
add("Wide Orange Yellow", band(600, 170, .75))
add("Violet Blue Long", band(445, 40, .45, 8))

rows = []
for i, (nm, fn) in enumerate(gel, 1):
    v = [round(max(0.0, min(1.0, fn(l))) * 100, 1) for l in W]
    rows.append([f"gel-{i:03d}", nm, v])
out = {"w0": 380, "s": 5, "f": rows}
p = Path(__file__).resolve().parent.parent / "data" / "gels.json"
p.write_text(json.dumps(out, separators=(",", ":")), encoding="utf-8")
print(len(rows), "gels ->", p)

#!/usr/bin/env python3
"""Draw the app icons (src/pwa/icons/*.png): the CIE spectral-locus horseshoe on a teal tile. Needs Pillow."""
import re
from pathlib import Path
from PIL import Image, ImageDraw
ROOT = Path(__file__).resolve().parent.parent
tab = re.search(r"const CMF_TAB=(\[\[.*?\]\]);", (ROOT / "src/js/core/grid-cmf.js").read_text(), re.S).group(1)
cmf = eval(tab)
pts = []
for x_, y_, z_ in cmf[20:321:4]:          # 400-700 nm
    s = x_ + y_ + z_
    pts.append((x_ / s, y_ / s))
S = 4  # supersampling

def draw(size, pad_frac, rounded):
    n = size * S
    im = Image.new("RGBA", (n, n), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    bg = (11, 125, 138, 255)
    if rounded: d.rounded_rectangle([0, 0, n - 1, n - 1], radius=int(n * .22), fill=bg)
    else: d.rectangle([0, 0, n, n], fill=bg)
    lo_x, hi_x, lo_y, hi_y = 0.0, 0.75, 0.0, 0.85
    box = n * (1 - 2 * pad_frac)
    def P(x, y): return (n * pad_frac + (x - lo_x) / (hi_x - lo_x) * box, n * pad_frac + (1 - (y - lo_y) / (hi_y - lo_y)) * box)
    poly = [P(x, y) for x, y in pts]
    d.polygon(poly, fill=(255, 255, 255, 235))
    d.line(poly + [poly[0]], fill=(255, 255, 255, 255), width=max(2, n // 60), joint="curve")
    cx, cy = P(0.3127, 0.329); r = n * .045
    d.ellipse([cx - r, cy - r, cx + r, cy + r], fill=bg)
    return im.resize((size, size), Image.LANCZOS)

out = ROOT / "src/pwa/icons"; out.mkdir(exist_ok=True)
draw(192, .16, True).save(out / "icon-192.png")
draw(512, .16, True).save(out / "icon-512.png")
draw(512, .26, False).save(out / "icon-maskable-512.png")      # full-bleed, extra padding for the safe zone
print("icons ->", out)

===== tools/spectra_folder_to_json.py =====
#!/usr/bin/env python3
"""Turn a folder of two-column spectra files (wavelength, value) into a data-set JSON for data/.

    python3 tools/spectra_folder_to_json.py my_pigments/ -o data/pigments.json
    python3 tools/spectra_folder_to_json.py my_pigments/ -o data/pigments.json --tidy-pigment-names --names names.json

Input files: .txt/.csv/.tsv, any of tab, space, semicolon or comma between columns, comma or dot decimals.
The value column is taken as percent (0-100); use --fraction if it is 0-1.
Output (percent, rounded to 0.1, on a 1 nm grid):  {"w0":380,"s":1,"f":[[id, name, [values...]], ...]}
Spectra must cover 380-780 nm (use --min-wl / --max-wl to relax that, the ends are then held).
"""
import argparse, json, os, re, sys
from pathlib import Path
import numpy as np

EXPAND = {}   # optional {"file stem": "Display name"} from --names


def tidy_pigment(stem):
    """'PR 264 - pyrrole red' -> 'Pyrrole red (PR 264)'"""
    m = re.match(r"^([A-Z][A-Za-z]{1,2}\s\d+)\s-\s(.+)$", stem)
    code, name = (m.group(1), m.group(2)) if m else ("", stem)
    name = name[0].upper() + name[1:]
    return f"{name} ({code})" if code else name


def read_spectrum(path):
    rows = []
    for line in open(path, errors="replace"):
        parts = [p for p in re.split(r"[\s;]+", line.strip().replace(",", ".")) if p]
        if len(parts) >= 2:
            try:
                rows.append((float(parts[0]), float(parts[1])))
            except ValueError:
                pass          # header or junk line
    a = np.array(rows)
    return a[np.argsort(a[:, 0])]


def to_grid(a, lo, hi):
    out = []
    for l in range(380, 781):
        s = (a[:, 0] >= l - .5) & (a[:, 0] < l + .5)          # bin-average if the data is denser than 1 nm
        out.append(a[s, 1].mean() if s.any() else np.interp(l, a[:, 0], a[:, 1]))
    return np.array(out)


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("folder"); ap.add_argument("-o", "--out", required=True)
    ap.add_argument("--fraction", action="store_true", help="values are 0-1 instead of percent")
    ap.add_argument("--tidy-pigment-names", action="store_true")
    ap.add_argument("--names", help="JSON {file stem: display name}")
    ap.add_argument("--min-wl", type=float, default=380.5); ap.add_argument("--max-wl", type=float, default=779.5)
    args = ap.parse_args()
    names = json.load(open(args.names)) if args.names else {}
    files = sorted([p for p in Path(args.folder).iterdir() if p.suffix.lower() in (".txt", ".csv", ".tsv")], key=lambda p: p.name.lower())
    out, seen = [], set()
    for p in files:
        a = read_spectrum(p)
        if len(a) < 5: print("skipped (no data):", p.name); continue
        if a[0, 0] > args.min_wl or a[-1, 0] < args.max_wl:
            sys.exit(f"{p.name}: covers {a[0,0]:g}-{a[-1,0]:g} nm, need 380-780 (relax with --min-wl/--max-wl)")
        v = to_grid(a, 380, 780) * (100 if args.fraction else 1)
        stem = p.stem
        name = names.get(stem) or (tidy_pigment(stem) if args.tidy_pigment_names else stem)
        slug = re.sub(r"[^a-z0-9]+", "-", name.lower()).strip("-")
        assert slug not in seen, f"duplicate id {slug}"; seen.add(slug)
        v = np.round(v, 1)
        out.append([slug, name, [float(x) if x % 1 else int(x) for x in v]])
    Path(args.out).write_text(json.dumps({"w0": 380, "s": 1, "f": out}, separators=(",", ":")) + "\n")
    print(f"{len(out)} spectra -> {args.out} ({os.path.getsize(args.out)/1024:.0f} KB)")


if __name__ == "__main__":
    main()

===== tools/spectra_matrix_to_csv.py =====
#!/usr/bin/env python3
"""Turn a plain-text spectra matrix (e.g. the Munsell measurements from the University of Eastern Finland) into the
wide CSV this app imports: first column wavelength_nm, one column per sample.

    python3 tools/spectra_matrix_to_csv.py spectra.txt out.csv [--names names.txt] [--first 380] [--step 1] [--chips-in-rows]

The matrix is whitespace/comma separated numbers. Layout is detected: if the first column looks like wavelengths
(380..800, increasing) it is used; otherwise give --first/--step for the wavelength of row 1 and the step.
--chips-in-rows says each ROW is one sample (columns are wavelengths). --names is one sample name per line
(default: sample-0001, ...). Values are written as read; the app accepts 0-1 fractions or percent.
This was written without access to the real files; check the first lines of the result before importing."""
import argparse, csv, re, sys
from pathlib import Path

ap = argparse.ArgumentParser()
ap.add_argument("matrix"); ap.add_argument("out")
ap.add_argument("--names"); ap.add_argument("--first", type=float); ap.add_argument("--step", type=float, default=1.0)
ap.add_argument("--chips-in-rows", action="store_true")
a = ap.parse_args()

rows = []
for line in Path(a.matrix).read_text(errors="replace").splitlines():
    t = [x for x in re.split(r"[\s,;]+", line.strip()) if x]
    if not t or t[0].startswith("#"):
        continue
    try:
        rows.append([float(x) for x in t])
    except ValueError:
        continue                                           # header/text line
if not rows:
    sys.exit("no numeric rows found")
if a.chips_in_rows:
    rows = [list(c) for c in zip(*rows)]                   # now rows = wavelengths, columns = samples
col0 = [r[0] for r in rows]
is_wl = all(300 <= x <= 1100 for x in col0) and all(b > a_ for a_, b in zip(col0, col0[1:])) and len(rows[0]) > 1
if is_wl:
    wl = col0; data = [r[1:] for r in rows]
else:
    if a.first is None:
        sys.exit("first column is not wavelengths: pass --first <nm of row 1> (and --step)")
    wl = [a.first + i * a.step for i in range(len(rows))]; data = rows
n = len(data[0])
names = [l.strip() for l in Path(a.names).read_text().splitlines() if l.strip()] if a.names else []
if names and len(names) != n:
    print(f"warning: {len(names)} names for {n} samples; using generic names", file=sys.stderr); names = []
names = names or [f"sample-{i + 1:04d}" for i in range(n)]
with open(a.out, "w", newline="") as f:
    w = csv.writer(f); w.writerow(["wavelength_nm"] + names)
    for x, r in zip(wl, data):
        w.writerow([f"{x:g}"] + [f"{v:g}" for v in r])
print(f"{n} samples x {len(wl)} wavelengths ({wl[0]:g}-{wl[-1]:g} nm) -> {a.out}")
